-- − Zapytanie 1: wybierające jedynie imię i nazwisko diagnosty laboratoryjnego z tabeli Diagnosta posortowane malejąco według nazwiska  

SELECT imie, nazwisko
FROM Diagnosta
ORDER BY nazwisko DESC;

-- − Zapytanie 2: zliczające liczbę zleceń pogrupowanych po datach (bez uwzględnienia godziny). Wybierane są jedynie daty i liczby zleceń 

SELECT COUNT(*) as ilosc, DATE(data) as data_zlecenia
FROM Zlecenie
GROUP BY DATE(data);

-- − Zapytanie 3: wybierające jedynie daty zleceń dla pacjenta o nazwisku Nowicki. Należy posłużyć się relacją

SELECT DATE(data) as data_zlecenia
FROM Zlecenie
    INNER JOIN pacjent ON zlecenie.idpacjenta = pacjent.id
WHERE nazwisko = 'Nowicki';

--  − Zapytanie 4: wybierające jedynie imię, nazwisko oraz pesel pacjenta z tabeli Pacjent, datę z tabeli Zlecenie, datę i plik z tabeli Wynik. Należy posłużyć się relacją i odpowiednim rodzajem złączenia. Wybierane są wszystkie zlecenia niezależenie od tego czy jest im przypisany wynik 

SELECT imie, nazwisko, pesel, zlecenie.data as data_zlecenia, wynik.data as data_wyniku, plik
FROM pacjent
    INNER JOIN Zlecenie ON zlecenie.idpacjenta = pacjent.id
    LEFT JOIN wynik ON zlecenie.id = wynik.idzlecenia;
